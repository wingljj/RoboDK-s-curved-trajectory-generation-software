cmake_minimum_required(VERSION 3.20)
project(SurfacePathStudio VERSION 0.1.2 LANGUAGES CXX)
if(WIN32)
  enable_language(RC)
endif()
if(MSVC)
  # The build script supplies the installed cl diagnostic language's prefix,
  # with a consistent codepage for CMake/Ninja dependency scanning.
  if(DEFINED ENV{SURFACEPATH_MSVC_INCLUDE_PREFIX})
    set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "$ENV{SURFACEPATH_MSVC_INCLUDE_PREFIX}")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "$ENV{SURFACEPATH_MSVC_INCLUDE_PREFIX}")
  endif()
  string(REPLACE "/EHsc" "/EHa" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
include(CTest)

option(SURFACEPATH_BUILD_APP "Build the native desktop application" ON)
option(SURFACEPATH_WINDOWS7 "Build the Windows 7 SP1 x64 compatibility edition with Qt 5 / MSVC 2019" OFF)
if(SURFACEPATH_WINDOWS7)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows 7 compatibility builds require Windows x64")
  endif()
  if(NOT MSVC OR MSVC_VERSION LESS 1920 OR MSVC_VERSION GREATER_EQUAL 1930)
    message(FATAL_ERROR "Windows 7 compatibility builds require the MSVC 2019 v142 toolset")
  endif()
  add_compile_definitions(WINVER=0x0601 _WIN32_WINNT=0x0601)
endif()
set(SURFACEPATH_OCCT_INCLUDE_DIR "" CACHE PATH "OCCT headers for an SDK without a usable package config")
set(SURFACEPATH_OCCT_LIBRARY_DIR "" CACHE PATH "OCCT libraries for an SDK without a usable package config")
set(SURFACEPATH_OCCT_SOURCE_DIR "" CACHE PATH "Optional matching OCCT sources, to build missing TKOpenGl")
option(SURFACEPATH_OCCT_STATIC "The manually configured OCCT SDK is static" OFF)

find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Concurrent)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Concurrent)
if(SURFACEPATH_WINDOWS7 AND (NOT QT_VERSION_MAJOR EQUAL 5 OR Qt5Core_VERSION VERSION_LESS 5.15))
  message(FATAL_ERROR "Windows 7 compatibility builds require Qt 5.15; Qt 6 cannot target Windows 7")
endif()

add_library(surfacepath_occt INTERFACE)
if(SURFACEPATH_OCCT_INCLUDE_DIR AND SURFACEPATH_OCCT_LIBRARY_DIR)
  target_include_directories(surfacepath_occt SYSTEM INTERFACE "${SURFACEPATH_OCCT_INCLUDE_DIR}")
  set(_occt_libs TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo
      TKTopAlgo TKPrim TKBO TKShHealing TKMesh TKHLR TKService TKV3d
      TKCDF TKLCAF TKCAF TKVCAF TKXCAF TKXSBase TKSTEPBase TKSTEPAttr
      TKSTEP209 TKSTEP TKXDESTEP)
  foreach(_lib IN LISTS _occt_libs)
    find_library(SPS_${_lib} NAMES ${_lib} PATHS "${SURFACEPATH_OCCT_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(surfacepath_occt INTERFACE "${SPS_${_lib}}")
  endforeach()
  if(SURFACEPATH_OCCT_STATIC)
    target_compile_definitions(surfacepath_occt INTERFACE OCCT_STATIC_BUILD)
  endif()
  find_library(SPS_TKOpenGl NAMES TKOpenGl PATHS "${SURFACEPATH_OCCT_LIBRARY_DIR}" NO_DEFAULT_PATH)
  if(SPS_TKOpenGl)
    target_link_libraries(surfacepath_occt INTERFACE "${SPS_TKOpenGl}")
  elseif(SURFACEPATH_OCCT_SOURCE_DIR)
    # Reuse the existing geometry SDK; compile only the official missing viewer toolkit.
    file(GLOB _opengl_sources CONFIGURE_DEPENDS "${SURFACEPATH_OCCT_SOURCE_DIR}/src/OpenGl/*.cxx")
    if(NOT _opengl_sources)
      message(FATAL_ERROR "Matching OCCT source tree has no OpenGl sources")
    endif()
    add_library(surfacepath_occt_opengl STATIC ${_opengl_sources})
    set_target_properties(surfacepath_occt_opengl PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_include_directories(surfacepath_occt_opengl SYSTEM PUBLIC
      "${SURFACEPATH_OCCT_INCLUDE_DIR}" "${SURFACEPATH_OCCT_SOURCE_DIR}/src/OpenGl")
    target_compile_definitions(surfacepath_occt_opengl PUBLIC OCCT_STATIC_BUILD)
    target_link_libraries(surfacepath_occt_opengl PUBLIC "${SPS_TKernel}" "${SPS_TKMath}" "${SPS_TKService}")
    if(MSVC)
      target_compile_options(surfacepath_occt_opengl PRIVATE /utf-8 /EHa /wd4996)
    endif()
    target_link_libraries(surfacepath_occt INTERFACE surfacepath_occt_opengl)
  elseif(SURFACEPATH_BUILD_APP)
    message(FATAL_ERROR "TKOpenGl is missing. Supply a complete OCCT SDK or matching SURFACEPATH_OCCT_SOURCE_DIR")
  endif()
else()
  find_package(OpenCASCADE REQUIRED)
  target_include_directories(surfacepath_occt SYSTEM INTERFACE "${OpenCASCADE_INCLUDE_DIR}")
  if(OpenCASCADE_LIBRARY_DIR)
    target_link_directories(surfacepath_occt INTERFACE "${OpenCASCADE_LIBRARY_DIR}")
  endif()
  foreach(_module FoundationClasses ModelingData ModelingAlgorithms Visualization ApplicationFramework DataExchange)
    if(OpenCASCADE_${_module}_LIBRARIES)
      target_link_libraries(surfacepath_occt INTERFACE ${OpenCASCADE_${_module}_LIBRARIES})
    endif()
  endforeach()
  if(NOT OpenCASCADE_FoundationClasses_LIBRARIES)
    target_link_libraries(surfacepath_occt INTERFACE ${OpenCASCADE_LIBRARIES})
  endif()
endif()
if(WIN32)
  target_link_libraries(surfacepath_occt INTERFACE opengl32 glu32 user32 gdi32 advapi32 ws2_32)
endif()

file(GLOB _core_sources CONFIGURE_DEPENDS src/cad/*.cpp src/geometry/*.cpp src/path/*.cpp src/export/*.cpp)
add_library(surfacepath_core ${_core_sources})
target_include_directories(surfacepath_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(surfacepath_core PUBLIC surfacepath_occt Qt${QT_VERSION_MAJOR}::Core)
if(MSVC)
  target_compile_options(surfacepath_core PRIVATE /W4 /utf-8 /EHa)
else()
  target_compile_options(surfacepath_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
  add_subdirectory(tests)
endif()
if(SURFACEPATH_BUILD_APP)
  file(GLOB _app_sources CONFIGURE_DEPENDS src/app/*.cpp src/ui/*.cpp src/viewer/*.cpp)
  add_executable(SurfacePathStudio WIN32 src/main.cpp ${_app_sources} resources/branding.qrc)
  if(WIN32)
    target_sources(SurfacePathStudio PRIVATE resources/application.rc)
  endif()
  target_include_directories(SurfacePathStudio PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}/resources")
  target_compile_definitions(SurfacePathStudio PRIVATE SPS_APP_VERSION="${PROJECT_VERSION}")
  if(SURFACEPATH_WINDOWS7)
    target_compile_definitions(SurfacePathStudio PRIVATE SPS_WINDOWS7_COMPAT=1)
    target_link_options(SurfacePathStudio PRIVATE "/SUBSYSTEM:WINDOWS,6.01" "/OSVERSION:6.1")
  endif()
  target_link_libraries(SurfacePathStudio PRIVATE surfacepath_core Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Concurrent)
  if(TARGET surfacepath_examples)
    target_link_libraries(SurfacePathStudio PRIVATE surfacepath_examples)
    target_compile_definitions(SurfacePathStudio PRIVATE SURFACEPATH_EXAMPLES_AVAILABLE)
  endif()
  if(MSVC)
    target_compile_options(SurfacePathStudio PRIVATE /W4 /utf-8 /EHa)
  endif()
  install(TARGETS SurfacePathStudio RUNTIME DESTINATION .)
endif()
