cmake_minimum_required(VERSION 3.20)

# A small standalone entry point lets geometry acceptance tests run before
# the desktop viewer and its OpenGL toolkit have been built.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(SurfacePathGeometryTests LANGUAGES CXX)
    if(MSVC)
        set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "Note: including file: ")
    endif()
    enable_testing()
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_PREFIX_PATH "D:/Software/QT/content/6.5.3/msvc2019_64" CACHE PATH "Qt prefix")
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core)
    set(OCCT_BUILD_ROOT "E:/surfacepath-poc-20261002/.deps/occt-static-build" CACHE PATH "Existing OCCT static build")
    file(GLOB _geometry_sources CONFIGURE_DEPENDS ../src/geometry/*.cpp ../src/path/*.cpp ../src/cad/*.cpp ../src/export/*.cpp)
    add_library(surfacepath_core ${_geometry_sources})
    target_include_directories(surfacepath_core PUBLIC ../src "${OCCT_BUILD_ROOT}/inc")
    target_compile_definitions(surfacepath_core PUBLIC OCCT_STATIC_BUILD)
    file(GLOB _occt_libraries "${OCCT_BUILD_ROOT}/win64/vc14/lib/*.lib")
    target_link_libraries(surfacepath_core PUBLIC ${_occt_libraries} user32 gdi32 advapi32 winmm psapi ws2_32 Qt${QT_VERSION_MAJOR}::Core)
    if(MSVC)
        string(REPLACE "/EHsc" "/EHa" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
        target_compile_options(surfacepath_core PRIVATE /utf-8)
    endif()
endif()
